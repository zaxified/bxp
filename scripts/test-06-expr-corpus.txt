# ── REGEX_MATCH / REGEX_EXTRACT ───────────────────────────────────────
# Pike-VM regex (linear-time, ReDoS-safe). The validator compiles the literal
# pattern during eval, so a malformed pattern is caught here as a loud,
# pattern-attributed BadRegexPattern — same static-literal verdict as
# DATE_CONVERT's DateFormatBadToken. A valid pattern (even a classic
# catastrophic-backtracking shape like `(a+)+$`) is ok: the engine cannot
# explode, so it stays a clean compile.
expr	ok	REGEX_MATCH('AAPL US Equity', '[A-Z]+')
expr	ok	REGEX_MATCH([Symbol], '^[A-Z]{2}[A-Z0-9]{9}[0-9]$')
expr	ok	REGEX_EXTRACT('Qualified Dividend AAPL 100', '[A-Z]{2,}')
# Capture group: the first group's text wins over the whole match.
expr	ok	REGEX_EXTRACT('order id=42 qty=3', 'id=([0-9]+)')
# Unicode-scalar mode: `.` and class ranges span code points, so accented
# letters match via an explicit class like `[A-ZÁ-Ž]` (not `\w`, which is ASCII).
expr	ok	REGEX_MATCH('Škoda', '[A-ZÁ-Ž]')
expr	ok	REGEX_EXTRACT([Note], '[A-ZÁ-Ž]{2,}')
