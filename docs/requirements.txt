# Build-time docs toolchain (MkDocs static site). Not a runtime dependency of
# bxp itself — install into a local, gitignored virtualenv:
#
#   python3 -m venv .venv-docs
#   .venv-docs/bin/pip install -r docs/requirements.txt
#   .venv-docs/bin/mkdocs serve      # live preview on http://127.0.0.1:8000
#   .venv-docs/bin/mkdocs build --strict
#
